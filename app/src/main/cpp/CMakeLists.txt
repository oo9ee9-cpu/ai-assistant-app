cmake_minimum_required(VERSION 3.22.1)
project(AsterEngine LANGUAGES CXX)
if(ANDROID)
 add_library(aster_engine SHARED EngineRuntime.cpp OpenGLES3Renderer.cpp NativeBridge.cpp)
 target_include_directories(aster_engine PRIVATE include)
 target_compile_features(aster_engine PRIVATE cxx_std_20)
 target_compile_options(aster_engine PRIVATE -Wall -Wextra -Wpedantic)
 find_library(GLES_LIBRARY GLESv3)
 find_library(LOG_LIBRARY log)
 target_link_libraries(aster_engine PRIVATE ${GLES_LIBRARY} ${LOG_LIBRARY})
else()
 add_library(aster_core STATIC EngineRuntime.cpp)
 target_include_directories(aster_core PUBLIC include)
 target_compile_features(aster_core PUBLIC cxx_std_20)
 target_compile_options(aster_core PRIVATE -Wall -Wextra -Wpedantic)
 option(ASTER_BUILD_TESTS "Build core tests" ON)
 if(ASTER_BUILD_TESTS)
  enable_testing()
  add_executable(aster_core_tests tests/CoreTests.cpp)
  target_link_libraries(aster_core_tests PRIVATE aster_core)
  add_test(NAME aster_core_tests COMMAND aster_core_tests)
 endif()
endif()
